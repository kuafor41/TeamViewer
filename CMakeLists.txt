cmake_minimum_required(VERSION 3.20)

project(Acrobat VERSION 0.2.0 LANGUAGES CXX)

option(ACROBAT_BUILD_TESTS "Build the core unit tests" ON)

add_library(AcrobatCore src/motion.cpp src/scheduler.cpp)
target_include_directories(AcrobatCore PUBLIC include)
target_compile_features(AcrobatCore PUBLIC cxx_std_20)

if(MSVC)
    target_compile_options(AcrobatCore PRIVATE /W4 /permissive- /EHsc /utf-8)
else()
    target_compile_options(AcrobatCore PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(WIN32)
    add_executable(Acrobat WIN32 src/win32_main.cpp)
    target_link_libraries(Acrobat PRIVATE AcrobatCore user32 shell32 advapi32 comctl32)
    target_compile_definitions(Acrobat PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
    if(MSVC)
        target_compile_options(Acrobat PRIVATE /W4 /permissive- /EHsc /utf-8)
    endif()
else()
    message(STATUS "Acrobat GUI target is Windows-only; core and tests remain portable.")
endif()

if(ACROBAT_BUILD_TESTS)
    enable_testing()
    add_executable(AcrobatCoreTests tests/core_tests.cpp)
    target_link_libraries(AcrobatCoreTests PRIVATE AcrobatCore)
    target_compile_features(AcrobatCoreTests PRIVATE cxx_std_20)
    if(MSVC)
        target_compile_options(AcrobatCoreTests PRIVATE /W4 /permissive- /EHsc /utf-8)
    else()
        target_compile_options(AcrobatCoreTests PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME AcrobatCoreTests COMMAND AcrobatCoreTests)
endif()
